1. git status (check repository)
2. git init (creat a new git repository)
3. git add --a OR git add . (stages all the changes)(all file)
4. git commit -m "message type(creat repository)"
5. git log (displays commited snapshot)(check auther and date)
6. rm -rf .git (delete repository)
7. git clone link example(https://github.com/tensorflow/tensorflow.git) (copy any repository)
8. pwd (check the current path)
9. ls (check all file in current path)
10. cd folder(tensorflow) (change the path in current folder)
11. cd .. (go back previous folder)
12. touch filename(.gitignore) (creat a new file)
13. .gitignore file to ignore any which don't want to commit (add filename in this file and folder to ignore)
14. git diff (there is change in the file OR see the changes)
15. git diff --staged (to check already stage changes)
16. git commit -a -m (skip staging area and commit)
17. git rm filename(third.txt) (delete the file in project)
18. git mv filename(a.txt) changed filename(b.txt)
19. git rm --cached filename(a.txt) (untracked the file)
20. git log -p (show the changes with commit)
21. git log --stat (show the all changed file with commit)
22. git log --pretty=oneline (commit show in oneline)
23. git log --pretty=short (show the author and msg)
24. git log --pretty=full ( show the commit, auther and msg)
25. git log --since=2.days (show 2 days commit) (use this:- days, months, years)
26. git log --pretty=formet:"%h -- %an" ("%h" = show abbreviated commit hash and "%an" = show author name)
27. git log -p -1 (show last commit with changed)
28. q and press enter (exit to log)
29. git commit --amend (show the editor of last one commit and you can change the msg click esc and write :wq change the msg)
30. git restore --staged filename(a.txt) (do the stage file to untracked file)
31. git checkout -- filename(a.txt) (restore file to previous commit)
32. git checkout -f (restore all file to previous commit)
33. git remote add (origin https://github.com/Bhargav9909/crud-in-php.git) (linkname and link)
34. git remote (give your linkname)
35. git remote -v (show the listed permission)
36. git push -u origin master (push the data in reposatory)(if the ssh key is accepted)
37. git pull origin master (pull/fetch the changed data in reposatory)(if the ssh key is accepted)
38. git config --global alias.st status (change status command to st)
39. git config --gloabal alias.unstage 'restore --staged --' (changed the command as unstage)
40. git checkout -b branchname(devlope) (create a new branch)
41. git checkout master (return to master branch and there folder)
42. git branch (to check how many branches are their)
43. git merge branchname(devlope) (merge the child branches to master branch)
44. git branch -d branchname(devlope) (delete the branch) (if the merge is not complete then give the error)
45. git branch -D branchname(devlope) (delete the branch without error)
46. git branch -v (show all the branch and the last commit with hash)
47. git branch --merged (show aleready merged branchs)
48. git branch --no-merged (show not merged branchs)
49. git push origin branchname(devlope):new branchname(fixed) (push data with new branchname without creating new branch)
50. git push -d origin branchname(devlope) (delete the on reposatory)
